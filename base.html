seed sample routes only if empty cur.execute("SELECT COUNT(*) AS c FROM routes") if cur.fetchone()["c"] == 0: sample = [ ("bus", "Saikrishnan Travels", "Chennai", "Bengaluru", "22:00", "05:30", 350, 40), ("bus", "Saikrishnan Travels", "Chennai", "Madurai", "21:30", "04:00", 460, 35), ("bus", "City Express", "Bengaluru", "Hyderabad", "20:00", "07:00", 570, 30), ("train", "Southern Railways", "Chennai", "Coimbatore", "06:00", "13:30", 500, 80), ("train", "Southern Railways", "Chennai", "Bengaluru", "23:00", "06:30", 360, 100), ("train", "Southern Railways", "Bengaluru", "Mumbai", "14:00", "10:00", 980, 120), ] cur.executemany( "INSERT INTO routes (mode, operator, source, destination, departure_time, " "arrival_time, distance_km, seats_available) VALUES (?,?,?,?,?,?,?,?)", sample, ) conn.commit() conn.close() # ------------------------------------------------------------- JAVA CLIENT -- def call_java_fare_service(distance_km: int, mode: str, seats: int): """ Calls the Java microservice to calculate fare and generate a ticket ID. Falls back to a local calculation if the Java service is unreachable, so the site keeps working even if that service is down. """ try: resp = requests.post( f"{JAVA_SERVICE_URL}/api/fare", json={"distanceKm": distance_km, "mode": mode, "seats": seats}, timeout=3, ) if resp.ok: data = resp.json() return data["fare"], data["ticketId"] except requests.exceptions.RequestException: pass # Fallback (local) calculation, mirrors the Java service logic rate = 2.5 if mode == "bus" else 1.8 fare = round(distance_km * rate * seats, 2) ticket_id = f"TT-{uuid.uuid4().hex[:8].upper()}" return fare, ticket_id # WHERE id = ?", (seats, route_id), ) conn.commit() conn.close() return redirect(url_for("confirmation", ticket_id=ticket_id)) conn.close() return render_template("book.html", route=route, owner=SITE_OWNER) @app.route("/confirmation/") def confirmation(ticket_id): conn = get_db() booking = conn.execute( "SELECT b.*, r.mode, r.operator, r.source, r.destination, r.departure_time, " "r.arrival_time FROM bookings b JOIN routes r ON b.route_id = r.id " "WHERE b.ticket_id = ?", (ticket_id,), ).fetchone() conn.close() if booking is None: flash("Ticket not found.") return redirect(url_for("home")) return render_template("confirmation.html", booking=booking, owner=SITE_OWNER) @app.route("/my-tickets", methods=["GET", "POST"]) def my_tickets(): bookings = [] email = "" if request.method == "POST": email = request.form.get("email", "").strip() conn = get_db() bookings = conn.execute( "SELECT b.*, r.mode, r.operator, r.source, r.destination, r.departure_time " "FROM bookings b JOIN routes r ON b.route_id = r.id " "WHERE b.passenger_email = ?
